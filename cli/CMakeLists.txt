cmake_minimum_required(VERSION 3.24)
project(MHARKitExport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(WIN32)
  # Trio's FileStream selects its UTF-8-to-wide path conversion with UNICODE.
  # Apply this to the static upstream libraries as well as the executable.
  add_compile_definitions(UNICODE _UNICODE)
endif()
set(OPENRIGLOGIC_SOURCE "" CACHE PATH "Clean checkout of cli/runtime.json's pinned OpenRigLogic revision")
if(NOT EXISTS "${OPENRIGLOGIC_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Set OPENRIGLOGIC_SOURCE to the pinned OpenRigLogic checkout")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/runtime.json" runtime_json)
string(JSON runtime_commit GET "${runtime_json}" commit)
execute_process(COMMAND git -C "${OPENRIGLOGIC_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE actual_commit OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git -C "${OPENRIGLOGIC_SOURCE}" status --porcelain OUTPUT_VARIABLE source_changes COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual_commit STREQUAL runtime_commit OR NOT source_changes STREQUAL "")
  message(FATAL_ERROR "OpenRigLogic must be a clean checkout of ${runtime_commit}")
endif()
set(READ_ONLY_SOURCE_TREE ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(RL_LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
set(RL_BUILD_PYTHON_WRAPPER "" CACHE STRING "" FORCE)
foreach(feature TESTS BENCHMARKS EXAMPLES WITH_AVX WITH_SSE WITH_NEON WITH_HALF_FLOATS)
  set(RL_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${OPENRIGLOGIC_SOURCE}" openriglogic EXCLUDE_FROM_ALL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" app_version)
string(STRIP "${app_version}" app_version)
if(MHARKIT_BUILD_INFO)
  file(READ "${MHARKIT_BUILD_INFO}" build_info)
else()
  set(build_info "{\"schema\":1,\"application\":\"mh-arkit-export\",\"version\":\"${app_version}\",\"implementation\":\"c++\",\"openriglogic\":${runtime_json}}")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/default_mapping.json" default_mapping)
configure_file(native/config.h.in generated/config.h @ONLY)
add_executable(mh-arkit-export native/main.cpp native/baker.cpp native/exports.cpp native/self_test.cpp)
target_include_directories(mh-arkit-export PRIVATE native "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(mh-arkit-export PRIVATE RigLogic::riglogic4)
if(MSVC)
  target_compile_options(mh-arkit-export PRIVATE /W4 /utf-8)
else()
  target_compile_options(mh-arkit-export PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(mh-arkit-export PRIVATE -static-libstdc++ -static-libgcc)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME native-self-test COMMAND mh-arkit-export --self-test)
endif()
install(TARGETS mh-arkit-export RUNTIME DESTINATION bin)
if(WIN32)
  target_link_libraries(mh-arkit-export PRIVATE shell32)
endif()
